cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(FrameCap VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(unofficial-libvpx CONFIG REQUIRED)
add_executable(FrameCap WIN32 src/main.cpp src/app.rc)
target_compile_definitions(FrameCap PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(FrameCap PRIVATE unofficial::libvpx::libvpx user32 gdi32 comdlg32 shell32)
if(MSVC)
    target_compile_options(FrameCap PRIVATE /W4 /permissive- /EHsc /utf-8)
    target_link_options(FrameCap PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
endif()
if(MINGW)
    target_compile_definitions(FrameCap PRIVATE FRAMECAP_EMBED_RC_MANIFEST)
    target_link_options(FrameCap PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
